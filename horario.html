// ════════════════════════════════════════════════════════════════ // HORARIOS OPERACIONES — Apps Script // ════════════════════════════════════════════════════════════════ const SHEET_ID_H = "11ZXfVvR9u1085RzOwg5JBO9YBIGiiufduGXtKRmv9Gw"; const HOJA_CONFIG = "Config"; const HOJA_HORARIOS = "Horarios"; const HOJA_ACCESOS = "Accesos"; const COL_NOMBRE = 0; const COL_CEDULA = 1; const COL_TIPO = 2; const COL_MODALIDAD = 3; const COL_ACTIVO = 4; const COL_TERRITORIO = 5; const COL_ROL = 6; // Columnas de la hoja "Accesos": Nombre | Correo | Clave | Territorio | Activo const COL_ACC_NOMBRE = 0; const COL_ACC_CORREO = 1; const COL_ACC_CLAVE = 2; const COL_ACC_TERRITORIO = 3; const COL_ACC_ACTIVO = 4; // ── NORMALIZAR FECHA ────────────────────────────────────────── function normFecha(val) { if (!val) return ""; if (val instanceof Date) { return String(val.getDate()).padStart(2,"0") + "/" + String(val.getMonth()+1).padStart(2,"0") + "/" + val.getFullYear(); } return String(val).trim(); } // ── NORMALIZAR HORA ─────────────────────────────────────────── function normHora(val) { if (!val && val !== 0) return ""; // Número decimal = fracción de día de Sheets (ej: 7/24 = 07:00) if (typeof val === "number") { const totalMin = Math.round(val * 24 * 60); const h = Math.floor(totalMin / 60) % 24; const m = totalMin % 60; return String(h).padStart(2,"0") + ":" + String(m).padStart(2,"0"); } if (val instanceof Date) { // OJO: no usar val.getTime() % 86400000 aquí — para fechas ancla de // Sheets (30/12/1899, usada en celdas de solo-hora), Bogotá tiene un // offset histórico de -4:56:16 (no -5:00) en la base de datos de zonas // horarias, así que esa cuenta desfasa la hora casi 5 minutos... y si // el proyecto de Apps Script tiene otro huso horario configurado, el // desfase real observado puede ser de varias horas (por eso 08:00 // aparecía como 12:56). getHours()/getMinutes() usan el mismo huso // horario del proyecto con el que Sheets generó el valor, así que // recuperan la hora local correcta sin importar el offset exacto. const h = val.getHours(); const m = val.getMinutes(); return String(h).padStart(2,"0") + ":" + String(m).padStart(2,"0"); } const s = String(val).trim(); const match = s.match(/^(\d{1,2}):(\d{2})/); if (match) return String(parseInt(match[1])).padStart(2,"0") + ":" + match[2]; return s; } // ── GET ─────────────────────────────────────────────────────── function doGet(e) { const action = (e.parameter.action || "").trim(); if (action === "getConfig") return getConfig(); if (action === "getHorarioColaborador") return getHorarioColaborador(e); return resp({ ok: false, error: "Acción no reconocida: " + action }); } // ── POST ────────────────────────────────────────────────────── function doPost(e) { try { const datos = JSON.parse(e.postData.contents); if (datos.action === "saveHorarios") return saveHorarios(datos); if (datos.action === "verificarAcceso") return verificarAcceso(datos.correo, datos.clave); return resp({ ok: false, error: "Acción no reconocida" }); } catch(err) { return resp({ ok: false, error: err.message }); } } // ── 0. VERIFICAR ACCESO (login del editor de horarios) ──────── // Hoja "Accesos": Nombre | Correo | Clave | Territorio | Activo function verificarAcceso(correo, clave) { try { const correoNorm = String(correo || "").trim().toLowerCase(); const claveNorm = String(clave || "").trim(); if (!correoNorm || !claveNorm) { return resp({ ok: false, error: "Ingresa correo y clave" }); } const ss = SpreadsheetApp.openById(SHEET_ID_H); const hAcc = ss.getSheetByName(HOJA_ACCESOS); if (!hAcc) return resp({ ok: false, error: "Hoja Accesos no encontrada" }); const filas = hAcc.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { const fCorreo = String(filas[i][COL_ACC_CORREO]).trim().toLowerCase(); if (fCorreo !== correoNorm) continue; const nombre = String(filas[i][COL_ACC_NOMBRE]).trim(); const fClave = String(filas[i][COL_ACC_CLAVE]).trim(); const territorio = String(filas[i][COL_ACC_TERRITORIO]).trim(); const activo = String(filas[i][COL_ACC_ACTIVO]).trim().toUpperCase(); if (activo !== "SI") return resp({ ok: false, error: "Este usuario está inactivo" }); if (fClave !== claveNorm) return resp({ ok: false, error: "Clave incorrecta" }); return resp({ ok: true, nombre: nombre, territorio: territorio }); } return resp({ ok: false, error: "Correo no encontrado" }); } catch(err) { return resp({ ok: false, error: err.message }); } } // Versión interna (booleana, sin envolver en resp) — para usar dentro de saveHorarios function esAccesoValido(correo, clave) { try { const correoNorm = String(correo || "").trim().toLowerCase(); const claveNorm = String(clave || "").trim(); if (!correoNorm || !claveNorm) return false; const ss = SpreadsheetApp.openById(SHEET_ID_H); const hAcc = ss.getSheetByName(HOJA_ACCESOS); if (!hAcc) return false; const filas = hAcc.getDataRange().getValues(); for (let i = 1; i < filas.length; i++) { const fCorreo = String(filas[i][COL_ACC_CORREO]).trim().toLowerCase(); if (fCorreo !== correoNorm) continue; const fClave = String(filas[i][COL_ACC_CLAVE]).trim(); const activo = String(filas[i][COL_ACC_ACTIVO]).trim().toUpperCase(); return activo === "SI" && fClave === claveNorm; } return false; } catch(e) { return false; } } // ── 1. OBTENER CONFIG ───────────────────────────────────────── function getConfig() { try { const ss = SpreadsheetApp.openById(SHEET_ID_H); const hConfig = ss.getSheetByName(HOJA_CONFIG); if (!hConfig) return resp({ ok: false, error: "Hoja Config no encontrada" }); const filas = hConfig.getDataRange().getValues(); const empleados = []; let filaInicio = 0; for (let i = 0; i < filas.length; i++) { if (String(filas[i][0]).toLowerCase().includes("nombre")) { filaInicio = i + 1; break; } } for (let i = filaInicio; i < filas.length; i++) { const nombre = String(filas[i][COL_NOMBRE]).trim(); const cedula = String(filas[i][COL_CEDULA]).trim(); const tipo = String(filas[i][COL_TIPO]).trim(); const modalidad = String(filas[i][COL_MODALIDAD]).trim(); const activo = String(filas[i][COL_ACTIVO]).trim().toUpperCase(); const rol = String(filas[i][COL_ROL]).trim(); if (!nombre || !cedula || activo === "NO") continue; empleados.push({ nombre, cedula, tipo, modalidad, activo, rol }); } const horarios = cargarTodosHorarios(); return resp({ ok: true, empleados, horarios }); } catch(err) { return resp({ ok: false, error: err.message }); } } // ── 2. CARGAR TODOS LOS HORARIOS ───────────────────────────── // ÚLTIMA ocurrencia gana (el editor siempre agrega al final) function cargarTodosHorarios() { try { const ss = SpreadsheetApp.openById(SHEET_ID_H); let hHor = ss.getSheetByName(HOJA_HORARIOS); if (!hHor) hHor = crearHojaHorarios(ss); const filas = hHor.getDataRange().getValues(); const result = {}; for (let i = 1; i < filas.length; i++) { const ced = String(filas[i][0]).trim(); const fecha = normFecha(filas[i][1]); if (!ced || !fecha) continue; if (!result[ced]) result[ced] = {}; // ÚLTIMA ocurrencia sobreescribe — el editor agrega al final result[ced][fecha] = { tipo: String(filas[i][2]).trim().toLowerCase() || "sin", entrada: normHora(filas[i][3]), salida: normHora(filas[i][4]), punto: String(filas[i][5] || "").trim() }; } return result; } catch(e) { Logger.log("Error: " + e.message); return {}; } } // ── 3. GUARDAR CAMBIOS ──────────────────────────────────────── function saveHorarios(datos) { try { // Solo personal autorizado (hoja "Accesos") puede guardar cambios if (!esAccesoValido(datos.correo, datos.clave)) { return resp({ ok: false, error: "No autorizado. Verifica tu correo y clave e ingresa de nuevo." }); } const ss = SpreadsheetApp.openById(SHEET_ID_H); let hHor = ss.getSheetByName(HOJA_HORARIOS); if (!hHor) hHor = crearHojaHorarios(ss); const cambios = datos.cambios || {}; const filas = hHor.getDataRange().getValues(); Object.keys(cambios).forEach(cedula => { Object.keys(cambios[cedula]).forEach(fecha => { const dia = cambios[cedula][fecha]; const tipo = dia.tipo || "sin"; const ent = dia.entrada || ""; const sal = dia.salida || ""; const punto = dia.punto || ""; // Buscar última fila existente para esa cédula+fecha let filaExistente = -1; for (let i = 1; i < filas.length; i++) { const fCed = String(filas[i][0]).trim(); const fFecha = normFecha(filas[i][1]); if (fCed === cedula && fFecha === fecha) { filaExistente = i + 1; // tomar la última } } if (filaExistente > 0) { hHor.getRange(filaExistente, 3, 1, 4) .setValues([[tipo, ent, sal, punto]]); } else { // OJO: nunca usar appendRow() directo con la fecha como texto — // Sheets la auto-convierte a un valor de fecha real y, cuando // llega por la API (a diferencia de cuando se escribe a mano en // la hoja), la interpreta como MM/DD/AAAA en vez de DD/MM/AAAA. // Eso hace que cualquier fecha con día 1-12 quede guardada con // el día y el mes invertidos (ej: "01/09/2026" queda como 9 de // enero en vez de 1 de septiembre) — invisible a simple vista // porque el formato de columna la vuelve a mostrar parecido. // Forzamos texto plano en toda la fila ANTES de escribir para // que Sheets nunca intente adivinar el formato. const filaNueva = hHor.getLastRow() + 1; const rango = hHor.getRange(filaNueva, 1, 1, 6); rango.setNumberFormat("@"); rango.setValues([[cedula, fecha, tipo, ent, sal, punto]]); } }); }); return resp({ ok: true, guardados: Object.keys(cambios).length }); } catch(err) { return resp({ ok: false, error: err.message }); } } // ── 4. HORARIO DE UN COLABORADOR ────────────────────────────── function getHorarioColaborador(e) { try { const cedula = String(e.parameter.cedula || "").trim(); const fechaIni = e.parameter.fechaIni || ""; const fechaFin = e.parameter.fechaFin || ""; const ss = SpreadsheetApp.openById(SHEET_ID_H); const hHor = ss.getSheetByName(HOJA_HORARIOS); if (!hHor) return resp({ ok: true, horario: [] }); const filas = hHor.getDataRange().getValues(); const fechas = generarFechas(fechaIni, fechaFin); const mapa = {}; for (let i = 1; i < filas.length; i++) { const filaCed = String(filas[i][0]).trim(); if (filaCed !== cedula) continue; const fechaStr = normFecha(filas[i][1]); if (!fechaStr) continue; // ÚLTIMA ocurrencia sobreescribe mapa[fechaStr] = { tipo: String(filas[i][2]).trim().toLowerCase() || "sin", entrada: normHora(filas[i][3]), salida: normHora(filas[i][4]), punto: String(filas[i][5] || "").trim(), }; } const horario = fechas.map(f => ({ fecha: f, ...(mapa[f] || { tipo: "sin" }) })); return resp({ ok: true, horario }); } catch(err) { return resp({ ok: false, error: err.message }); } } // ── CREAR HOJA HORARIOS ─────────────────────────────────────── function crearHojaHorarios(ss) { const hoja = ss.insertSheet(HOJA_HORARIOS); hoja.getRange(1, 1, 1, 6).setValues([["Cedula","Fecha","Tipo","Entrada","Salida","Punto"]]); hoja.getRange(1, 1, 1, 6).setBackground("#FF441B").setFontColor("#FFFFFF").setFontWeight("bold"); // Columnas A y B siempre en texto plano, para que Sheets nunca intente // auto-convertir la cédula o la fecha (ver nota en saveHorarios()). hoja.getRange(1, 1, 1000, 2).setNumberFormat("@"); [130,110,90,80,80,200].forEach((w,i) => hoja.setColumnWidth(i+1, w)); return hoja; } // ── GENERAR FECHAS ──────────────────────────────────────────── function generarFechas(ini, fin) { const [dI,mI,aI] = ini.split("/").map(Number); const [dF,mF,aF] = fin.split("/").map(Number); const start = new Date(aI, mI-1, dI); const end = new Date(aF, mF-1, dF); const arr = []; const cur = new Date(start); while(cur <= end) { arr.push(String(cur.getDate()).padStart(2,"0")+"/"+String(cur.getMonth()+1).padStart(2,"0")+"/"+cur.getFullYear()); cur.setDate(cur.getDate()+1); } return arr; } // ── RESPUESTA JSON ──────────────────────────────────────────── function resp(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
